pipeline {
    agent any

    parameters {
        choice(name: 'STREAM_MINUTES', choices: ['1', '3', '5'], description: 'How long to stream logs')
    }

    environment {
        SERVER_HOST    = 'developer@10.10.3.83'
        PROJECT_DIR    = '~/email-intelli-widget'
        COMPOSE_FILE   = 'docker-compose.backend.yml'
        TAIL_LINES     = '200'
        STREAM_SECONDS = "${params.STREAM_MINUTES.toInteger() * 60}"
        SSH_CREDS_ID   = 'ssh-key'
    }

    stages {

        stage('Stream Email Analysis Backend Logs') {
            steps {
                sshagent([env.SSH_CREDS_ID]) {
                    sh """
                        echo "▶ Live logs for Email Analysis Backend (uvicorn + celery) on ${env.SERVER_HOST}..."
                        ssh -o StrictHostKeyChecking=no ${env.SERVER_HOST} "
                            cd ${env.PROJECT_DIR} &&
                            timeout ${env.STREAM_SECONDS} docker compose -f ${env.COMPOSE_FILE} logs -f --tail=${env.TAIL_LINES} || true
                        "
                    """
                }
            }
        }

    }

    post {
        always {
            echo "✅ Log stream ended for Email Analysis Backend."
        }
        failure {
            echo "❌ Failed — containers may not be running. Check: docker ps"
        }
    }
}
